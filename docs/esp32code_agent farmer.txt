#include <WiFi.h>
#include <HTTPClient.h>

// =================================
// Wi-Fi
// =================================

const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";

// =================================
// ThingSpeak
// =================================

const char* thingspeakAPIKey = "YOUR_WRITE_API_KEY";

const char* server = "http://api.thingspeak.com/update";

// =================================
// ESP32 UART2
// =================================

#define RXD2 16
#define TXD2 17

// =================================
// Sensor variables
// =================================

float temperature = 0;
float humidity = 0;
float soilMoisture = 0;

String receivedData = "";

// =================================
// ThingSpeak timer
// =================================

unsigned long lastUploadTime = 0;

const unsigned long uploadInterval = 15000;

// =================================
// Setup
// =================================

void setup() {

  Serial.begin(115200);

  // UNO → ESP32
  Serial2.begin(9600, SERIAL_8N1, RXD2, TXD2);

  Serial.println();
  Serial.println("==============================");
  Serial.println("ESP32 + ThingSpeak");
  Serial.println("==============================");

  // Connect Wi-Fi
  WiFi.begin(ssid, password);

  Serial.print("Connecting to WiFi");

  while (WiFi.status() != WL_CONNECTED) {

    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi connected!");

  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());

  Serial.println("Waiting for UNO data...");
}

// =================================
// Main loop
// =================================

void loop() {

  // =================================
  // Receive data from UNO
  // =================================

  while (Serial2.available()) {

    char c = Serial2.read();

    if (c == '\n') {

      receivedData.trim();

      if (receivedData.length() > 0) {

        Serial.print("Received from UNO: ");
        Serial.println(receivedData);

        // Expected format:
        // temperature,humidity,soil

        int comma1 = receivedData.indexOf(',');

        int comma2 = receivedData.indexOf(
          ',',
          comma1 + 1
        );

        if (comma1 > 0 && comma2 > comma1) {

          // Temperature
          String tempString =
            receivedData.substring(
              0,
              comma1
            );

          // Humidity
          String humidityString =
            receivedData.substring(
              comma1 + 1,
              comma2
            );

          // Soil moisture
          String soilString =
            receivedData.substring(
              comma2 + 1
            );

          temperature = tempString.toFloat();

          humidity = humidityString.toFloat();

          soilMoisture = soilString.toFloat();

          // Print values
          Serial.println("-------------------------");

          Serial.print("Temperature: ");
          Serial.print(temperature);
          Serial.println(" C");

          Serial.print("Humidity: ");
          Serial.print(humidity);
          Serial.println(" %");

          Serial.print("Soil Moisture: ");
          Serial.print(soilMoisture);
          Serial.println(" %");

          Serial.println("-------------------------");
        }
      }

      receivedData = "";
    }

    else {

      receivedData += c;
    }
  }

  // =================================
  // Upload every 15 seconds
  // =================================

  if (millis() - lastUploadTime >= uploadInterval) {

    if (temperature != 0 ||
        humidity != 0 ||
        soilMoisture != 0) {

      sendToThingSpeak();

      lastUploadTime = millis();
    }
  }
}

// =================================
// Send data to ThingSpeak
// =================================

void sendToThingSpeak() {

  if (WiFi.status() != WL_CONNECTED) {

    Serial.println("WiFi disconnected!");

    WiFi.reconnect();

    return;
  }

  HTTPClient http;

  String url =
    String(server) +
    "?api_key=" +
    thingspeakAPIKey +
    "&field1=" +
    String(temperature, 1) +
    "&field2=" +
    String(humidity, 1) +
    "&field3=" +
    String(soilMoisture, 1);

  Serial.println();
  Serial.println("Uploading to ThingSpeak...");

  Serial.println(url);

  http.begin(url);

  int httpResponseCode = http.GET();

  if (httpResponseCode > 0) {

    Serial.print("HTTP Response: ");
    Serial.println(httpResponseCode);

    String response =
      http.getString();

    Serial.print("ThingSpeak Entry ID: ");
    Serial.println(response);

    Serial.println(
      "Upload successful!"
    );

  }
  else {

    Serial.print(
      "Upload failed. Error: "
    );

    Serial.println(
      httpResponseCode
    );
  }

  http.end();

  Serial.println();
}
